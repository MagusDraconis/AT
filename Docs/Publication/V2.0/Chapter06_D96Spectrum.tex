introduce no new physics and no speculation.

\section{The Spectrum Structure}
\label{sec:spectrum-structure}

\begin{definition}[The D96 spectrum]
\label{def:d96}
The \emph{D96 spectrum} is the Laplacian eigenspectrum of the converged $N=96$ attractor:
the set of $96$ eigenvalues of the graph Laplacian, consisting of one zero eigenvalue (the
background mode) and $95$ positive eigenvalues (the positive modes)
\cite{QG155,QG295}.
\end{definition}

\begin{theorem}[The D96 spectrum is the derived output]
\label{thm:d96-derived}
The D96 spectrum is the derived output of the actualization process: the Laplacian
eigenspectrum of the converged $N=96$ attractor, and not a primitive input of the theory.
\end{theorem}

\begin{proof}
By Theorem~\ref{thm:spectrum-eigenspectrum} of Chapter~\ref{chap:spectrum}, the spectrum of
the theory is the Laplacian eigenspectrum of the converged $N=96$ network. By
Theorem~\ref{thm:spectrum-not-primitive} of the same chapter, the spectrum is not a
primitive but the inevitable output of the attractor. Hence the D96 spectrum is a derived
structure: an output of the actualization process, carrying no independent choice. This
presentation is consistent with the canonical architecture of the spectrum layer
\cite{QG295}.
\end{proof}

\begin{corollary}[No primitive interpretation]
\label{cor:d96-not-primitive}
The D96 spectrum admits no primitive interpretation: it is the emergent eigenspectrum of
the attractor, not an underivable input.
\end{corollary}

\begin{proof}
By Theorem~\ref{thm:d96-derived}, the spectrum is the derived output of the process. The
spectrum-necessity audit \cite{QG295} confirms this classification explicitly: the spectrum
carries no choice and presupposes the actualization process and its primitive Difference.
Hence no primitive interpretation of the D96 spectrum is consistent with the canonical
architecture.
\end{proof}

\section{The Zero Mode}
\label{sec:zero-mode}

\begin{definition}[Zero mode]
\label{def:zero-mode}
The \emph{zero mode} is the eigenvalue $\lambda_0 = 0$ of the graph Laplacian: the
background mode of the spectrum.
\end{definition}

\begin{theorem}[The zero mode is the background]
\label{thm:zero-mode}
The D96 spectrum has exactly one zero eigenvalue, $\lambda_0 = 0$, corresponding to the
background of the theory: the uniform configuration from which the count is measured.
\end{theorem}

\begin{proof}
The graph Laplacian of a connected network has exactly one zero eigenvalue, with the
constant eigenvector --- the uniform background. The spectrum-necessity audit \cite{QG295}
identifies this zero mode as the background. Hence the D96 spectrum has one zero mode, the
background against which the positive modes and the count structure are defined.
\end{proof}

\begin{corollary}[The background is the reference of Difference]
\label{cor:zero-background}
The zero mode is the uniform background of the count: the reference against which
Difference (the counting difference from a uniform background) is measured.
\end{corollary}

\begin{proof}
By Theorem~\ref{thm:zero-mode}, the zero mode is the uniform background configuration. By
Definition~\ref{def:difference} of Chapter~\ref{chap:difference}, Difference is the counting
difference from a uniform background. Hence the zero mode is the reference of the count ---
the uniform background from which the positive-mode structure represents the differences.
\end{proof}

\section{Positive Modes}
\label{sec:positive-modes}

\begin{definition}[Positive modes]
\label{def:positive-modes}
The \emph{positive modes} are the $95$ positive eigenvalues
$\lambda_1,\dots,\lambda_{95}$ of the graph Laplacian: the non-background degrees of
freedom of the spectrum.
\end{definition}

\begin{theorem}[There are 95 positive modes]
\label{thm:95-modes}
The D96 spectrum has exactly $95$ positive modes: the $96$-dimensional spectrum consists of
one zero mode and $95$ positive eigenvalues.
\end{theorem}

\begin{proof}
The graph Laplacian of the $N=96$ network is a $96\times96$ matrix with one zero eigenvalue
(Theorem~\ref{thm:zero-mode}) and the remaining $95$ eigenvalues positive. The
doublet-multiplicity structure of the spectrum confirms the count: $\Sigma m = 95$ is the
total mode count of the positive spectrum \cite{QG157}. Hence the D96 spectrum has exactly
$95$ positive modes.
\end{proof}

\begin{corollary}[95 modes, one background]
\label{cor:95-1}
The D96 spectrum consists of $95$ positive modes plus one zero mode: the total dimension
is $96$, with the background contributing the single zero eigenvalue.
\end{corollary}

\begin{proof}
By Theorem~\ref{thm:zero-mode} and Theorem~\ref{thm:95-modes}, the spectrum has one zero
mode and $95$ positive modes. Hence the total is $96$ modes, matching the $N=96$ dimension
of the attractor \cite{QG155}.
\end{proof}

\section{Spectral Moments}
\label{sec:spectral-moments}

\begin{definition}[Spectral moment]
\label{def:moment}
The \emph{$p$-moment} of the multiplicity distribution is
\begin{equation}
\label{eq:moment}
\Sigma m^p \;=\; \sum_i m_i^p,
\end{equation}
where the sum runs over the degenerate groups of the spectrum, with multiplicities $m_i$.
\end{definition}

\begin{theorem}[The first moment is the total mode count]
\label{thm:sum-m}
The first spectral moment is $\Sigma m = 95$: the total number of positive modes of the
D96 spectrum.
\end{theorem}

\begin{proof}
The effective-access-count program \cite{QG157} establishes that the first moment of the
doublet-multiplicity distribution is the total mode count. By
Theorem~\ref{thm:95-modes}, the total positive-mode count is $95$. Hence
$\Sigma m = \sum_i m_i = 95$, the full positive spectrum accessed uniformly.
\end{proof}

\begin{theorem}[The half-moment]
\label{thm:sum-sqrt-m}
The half-moment of the multiplicity distribution is
$\Sigma\sqrt{m} = 64.08$: the neutral-sector access count.
\end{theorem}

\begin{proof}
The effective-access-count program \cite{QG157} establishes that the half-moment
$\Sigma\sqrt{m} = \sum_i \sqrt{m_i}$ is the neutral access count: the statistical weight
each degenerate group contributes to the neutral channel. The numerical value
$\Sigma\sqrt{m} = 64.08$ follows from the multiplicity structure $[42\times2,5,6]$.
\end{proof}

\begin{theorem}[The second moment]
\label{thm:sum-m2}
The second spectral moment is $\Sigma m^2 = 229$: the doublet-occupancy access count.
\end{theorem}

\begin{proof}
The effective-access-count program \cite{QG157} establishes that the second moment
$\Sigma m^2 = \sum_i m_i^2$ is the doublet-occupancy access count. The numerical value
$\Sigma m^2 = 229$ follows from the multiplicity structure
$[42\times2,5,6]$: $42\cdot 2^2 + 5^2 + 6^2 = 168+25+36 = 229$.
\end{proof}

\begin{corollary}[The moment ladder]
\label{cor:moment-ladder}
The spectral moments form a ladder: $\Sigma\sqrt{m} = 64.08$ (half),
$\Sigma m = 95$ (first), $\Sigma m^2 = 229$ (second) --- each an access count of a
different sector of the theory \cite{QG157,QG158}.
\end{corollary}

\begin{proof}
By Theorem~\ref{thm:sum-sqrt-m}, Theorem~\ref{thm:sum-m}, and
Theorem~\ref{thm:sum-m2}, the three moments are the half, first, and second moments of the
same multiplicity distribution. The moment-orders program \cite{QG158} establishes that the
moments form a Z2-power ladder, each corresponding to a distinct sector access.
\end{proof}

\section{Spectral Constants}
\label{sec:spectral-constants}

\begin{definition}[Occupancy moment]
\label{def:occmom}
The \emph{octave-occupation moment} is
\begin{equation}
\label{eq:occmom}
\mathrm{occMom} \;=\; \frac{\sum_{\text{octaves}} \mathrm{occ}_j^2}{\mathrm{occ}_0},
\end{equation}
where $\mathrm{occ}_j$ are the octave occupancies and $\mathrm{occ}_0$ the lightest-octave
occupancy.
\end{definition}

\begin{theorem}[The occupancy moment]
\label{thm:occmom}
The octave-occupation moment of the D96 spectrum is
$\mathrm{occMom} = 1900.25$: the up-sector (dense-band) access count.
\end{theorem}

\begin{proof}
The effective-access-count program \cite{QG157} establishes that the octave-occupation
moment $\mathrm{occMom} = \sum_j \mathrm{occ}_j^2/\mathrm{occ}_0$ is the dense-band access
count of the up sector. The octave occupancies of the D96 spectrum --- the mode counts per
octave band --- yield $\mathrm{occMom} = 1900.25$.
\end{proof}

\begin{theorem}[The span]
\label{thm:span}
The span of the D96 spectrum is
\begin{equation}
\label{eq:span}
\mathrm{span} \;=\; \frac{\omega_{\max}}{\omega_{\min}} \;=\; 6.40,
\end{equation}
the ratio of the highest to the lowest positive frequency.
\end{theorem}

\begin{proof}
The span is the ratio of the extremal positive modes of the spectrum, $\omega_{\max}$ and
$\omega_{\min}$. The D96 spectrum yields the numerical value $\mathrm{span} = 6.40$,
established by the spectral-structure program \cite{QG155}. The span measures the frequency
extent of the spectrum, a derived structural constant.
\end{proof}

\begin{corollary}[All constants are derived]
\label{cor:constants-derived}
The spectral constants --- $\Sigma m=95$, $\Sigma\sqrt{m}=64.08$, $\Sigma m^2=229$,
$\mathrm{occMom}=1900.25$, $\mathrm{span}=6.40$ --- are all derived from the D96 spectrum,
which is itself derived from the attractor.
\end{corollary}

\begin{proof}
Each constant is a moment or structural feature of the spectrum
(Theorem~\ref{thm:sum-m}, Theorem~\ref{thm:sum-sqrt-m}, Theorem~\ref{thm:sum-m2},
Theorem~\ref{thm:occmom}, Theorem~\ref{thm:span}). By
Theorem~\ref{thm:d96-derived}, the spectrum is the derived output of the attractor. Hence
every constant is twice derived: a function of the spectrum, which is a function of the
process. No constant is a primitive input \cite{QG157,QG295}.
\end{proof}

\section{Multiplicity Structure}
\label{sec:multiplicity}

\begin{definition}[Multiplicity structure]
\label{def:multiplicity}
The \emph{multiplicity structure} of the D96 spectrum is the multiset of degenerate-group
sizes $m_i$: the pattern of equal eigenvalues grouped into the doublet structure.
\end{definition}

\begin{theorem}[The multiplicity structure]
\label{thm:multiplicity}
The multiplicity structure of the D96 spectrum is $[42\times2,\,5,\,6]$: forty-two
doublets, one group of size $5$, and one group of size $6$.
\end{theorem}

\begin{proof}
The doublet-multiplicity structure of the spectrum is established by the effective-access-
count program \cite{QG157}: the spectrum is a multiset of degenerate groups with
multiplicities $m_i$, consisting of $42$ groups of size $2$, one of size $5$, and one of
size $6$. The Z2 doublet structure is generated by the D96 symmetry \cite{QG155}, and the
total satisfies $\Sigma m = 42\cdot2 + 5 + 6 = 95$, consistent with
Theorem~\ref{thm:sum-m}.
\end{proof}

\begin{corollary}[The doublet symmetry]
\label{cor:doublet-symmetry}
The dominant multiplicity is the doublet: $42$ of the $44$ groups are of size $2$,
reflecting the Z2 pairing of the spectrum generated by the D96 symmetry.
\end{corollary}

\begin{proof}
By Theorem~\ref{thm:multiplicity}, the spectrum has $42$ doublets among $44$ groups. The
Z2 doublet structure is generated by the D96 symmetry \cite{QG155}: the observable-sector
spectrum is fully Z2 paired. Hence the doublet is the dominant structural unit of the
spectrum.
\end{proof}
